189	OB_INLINE bool ObTenantDfc::can_unblock(ObDtlFlowControl *dfc)
190	{
191	  bool can_unblock = false;
192	  // 如果有一个channel，一直接收比较大的msg，如block等，会不会导致全部block住，只有等待该sql运行完，其他session才会继续
193	  if (nullptr != dfc && dfc->is_block()) {
194	    // 这里有几种策略进行unblocking操作
195	    // 1 直接根据当前dfc的状态发送unblocking msg
196	    // 2 如果超卖，则根据全局使用情况发送unblock msg
197	    // 需要注意的是，如果既需要看当前dfc满足条件，又需要满足全局使用情况，则dfc_server当满足时，对所有dfc发送unblocking状况
198	    // 否则会导致某个channel由于之前达到了自身可以unblock条件，由于没有发送unblocking msg，后面可能没有该channel的任何msg处理，这样就永远再发送unblocking，deadlock
199	    // 所以必须dfc server在达到unblocking条件时，对当前所有的dfc通知unblocking，感觉这样有点重
200	    // 现仅仅满足其中之一就通知unblocking
201	    int64_t max_size_per_dfc = get_max_size_per_channel() * dfc->get_channel_count();
202	    can_unblock = dfc->can_unblock() ||
203	      (dfc->get_used() <= max_size_per_dfc / 2 && tenant_dfc_.get_used() < max_buffer_size_);
204	  }
205	  return can_unblock;
206	}
